Pages CMS Configuration (.pages.yml)

Place this file in your root GitHub repository as .pages.yml to enable Pages CMS admin controls for G-H Maker Studio. 

media:
  input: assets/images
  output: /assets/images

content:
  - name: homepage
    label: "Homepage Settings"
    type: file
    path: "data/homepage.json"
    fields:
      - { name: hero_title, label: "Hero Title", type: string }
      - { name: hero_subtitle, label: "Hero Subtitle", type: text }
      - { name: announcement_bar, label: "Announcement Text", type: string }

  - name: workshops
    label: "Workshops Schedule"
    type: collection
    path: "data/workshops"
    fields:
      - { name: title, label: "Workshop Title", type: string }
      - { name: date, label: "Date & Time", type: string }
      - { name: price, label: "Price ($)", type: number }
      - { name: seats_available, label: "Available Seats", type: number }

  - name: products
    label: "Handmade Products"
    type: collection
    path: "data/products"
    fields:
      - { name: name, label: "Product Name", type: string }
      - { name: price, label: "Price ($)", type: number }
      - { name: category, label: "Category", type: string }
      - { name: image, label: "Image", type: image }
